resolution_mygo_galactic_tour_planet_initiative = {
	icon = "GFX_resolution_community_champions"

	resources = {
		category = resolutions
		cost = {
			influence = @mygo_resolution_cost_t1
		}
	}
	target = no
	level = 1

	modifier = {
		resolution_mygo_galactic_tour_planet_initiative = 1
	}

	effect = {
		custom_tooltip = resolution_mygo_galactic_tour_planet_initiative_tooltip
		hidden_effect = {
			mygo_resolution_vote_passed_effect = yes
			every_playable_country = {
				limit = {
					is_galactic_community_member = yes
				}
				every_owned_planet = {
					planet = {
						if = {
							limit = {
								has_planet_flag = mygo_tour_visited
								NOT = { has_modifier = mygo_resolution_tour_planet_bonus }
							}
							add_modifier = {
								modifier = mygo_resolution_tour_planet_bonus
								days = -1
							}
						}
					}
				}
			}
		}
	}

	fail_effects = {
		hidden_effect = {
			mygo_resolution_vote_failed_effect = yes
		}
	}

	potential = {
		any_country = {
			has_origin = origin_mygo_eternal_return
			has_country_flag = mygo_tour_unlocked
			is_galactic_community_member = yes
		}
	}

	allow = {
		has_origin = origin_mygo_eternal_return
		has_country_flag = mygo_tour_unlocked
		custom_tooltip = {
			fail_text = resolution_already_active
			NOR = {
				is_active_resolution = "resolution_mygo_galactic_tour_planet_initiative"
				is_active_resolution = "resolution_mygo_galactic_tour_mutual_benefit"
				is_active_resolution = "resolution_mygo_galactic_tour_public_stage_fund"
				is_active_resolution = "resolution_mygo_galactic_tour_universal_mandate"
			}
		}
	}

	ai_weight = {
		base = @mygo_resolution_weight_loved
		modifier = {
			factor = 2
			is_xenophile = yes
			desc = ethic_xenophile
		}
		modifier = {
			factor = 1.5
			is_egalitarian = yes
			desc = ethic_egalitarian
		}
		modifier = {
			factor = 1.5
			is_megacorp = yes
			desc = auth_corporate
		}
		modifier = {
			factor = 0.5
			is_xenophobe = yes
			desc = ethic_xenophobe
		}
	}
}

		}
	}

	effect = {
		custom_tooltip = resolution_mygo_galactic_tour_universal_mandate_tooltip
		hidden_effect = {
			mygo_resolution_vote_passed_effect = yes
			set_global_flag = resolution_with_breach_effect_passed
			every_playable_country = {
				limit = {
					is_galactic_community_member = yes
					NOT = { has_origin = origin_mygo_eternal_return }
					NOT = { has_country_flag = mygo_tour_first_completed }
				}
				set_country_flag = mygo_tour_law_grace_assigned
				set_timed_country_flag = {
					flag = mygo_tour_law_grace
					days = 3600
				}
			}
		}
	}

	fail_effects = {
		hidden_effect = {
			mygo_resolution_vote_failed_effect = yes
		}
	}

	potential = {
		any_country = {
			has_origin = origin_mygo_eternal_return
			has_country_flag = mygo_tour_unlocked
			is_galactic_community_member = yes
		}
	}

	allow = {
		is_active_resolution = "resolution_mygo_galactic_tour_public_stage_fund"
	}

	breach = {
		NOT = { has_country_flag = constitutional_immunity }
		NOT = { has_origin = origin_mygo_eternal_return }
		NOT = { has_country_flag = mygo_tour_first_completed }
		NOT = { has_country_flag = mygo_tour_law_grace }
		has_country_flag = mygo_tour_law_grace_assigned
		any_owned_planet = {
			is_colony = yes
			is_under_colonization = no
		}
		any_country = {
			has_origin = origin_mygo_eternal_return
			has_country_flag = mygo_tour_unlocked
			is_galactic_community_member = yes
		}
	}

	ai_weight = {
		base = @mygo_resolution_weight_disliked
		modifier = {
			factor = 2
			has_country_flag = mygo_tour_first_completed
			desc = mygo_resolution_ai_completed_tour
		}
		modifier = {
			factor = 0.1
			NOT = { has_country_flag = mygo_tour_first_completed }
			desc = mygo_resolution_ai_not_completed_tour
		}
		modifier = {
			factor = 1.5
			is_xenophile = yes
			desc = ethic_xenophile
		}
		modifier = {
			factor = 0.5
			is_xenophobe = yes

		}
