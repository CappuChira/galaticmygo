special_project = {
	key = "MYGO_CONTRACT_COMPOSITION_PROJECT"
	event_chain = contract_mygo_composition_chain
	cost = 750
	tech_department = society_technology
	timelimit = -1
	picture = GFX_evt_society_research
	icon = "gfx/interface/icons/situation_log/situation_log_quest.dds"
	location = no
	remove_when_completed = yes

	event_scope = country_event

	on_success = {
		owner = {
			add_mission_counter = {
				mission = contract_mygo_composition
				counter = contract_progress_counter
				amount = 1
			}
		}
	}

	on_fail = {
		owner = { end_event_chain = contract_mygo_composition_chain }
	}

	abort_trigger = {
		NOT = {
			owner = { has_event_chain = contract_mygo_composition_chain }
		}
	}

	on_cancel = {
		# Do nothing
	}
}
