@mygo_parallel_homeworld_min_size = 20
@mygo_parallel_homeworld_max_size = 22

mygo_popipa_parallel_system = {
	class = "rl_starting_stars"
	usage = empire_init
	flags = { empire_home_system mygo_parallel_popipa_system }

	init_effect = {
		every_neighbor_system = {
			set_star_flag = empire_cluster
			every_neighbor_system = { set_star_flag = empire_cluster }
		}
		generate_home_system_resources = yes
	}

	planet = {
		count = 1
		class = star
		orbit_distance = 0
		orbit_angle = 1
		size = { min = 20 max = 30 }
	}

	change_orbit = 45

	planet = {
		count = 1
		class = pc_tropical
		home_planet = yes
		orbit_distance = 25
		orbit_angle = { min = 90 max = 270 }
		size = { min = @mygo_parallel_homeworld_min_size max = @mygo_parallel_homeworld_max_size }
		deposit_blockers = none
		modifiers = none
		flags = { mygo_parallel_popipa_homeworld }
		init_effect = {
			generate_empire_home_planet = yes
			prevent_anomaly = yes
		}
	}

	change_orbit = 25

	planet = {
		count = { min = 2 max = 4 }
		class = random_non_colonizable
		orbit_distance = 20
		orbit_angle = { min = 90 max = 270 }
	}

	neighbor_system = {
		trigger = { num_guaranteed_colonies >= 2 }
		hyperlane_jumps = { min = 1 max = 3 }
		initializer = "neighbor_t1"
	}
	neighbor_system = {
		trigger = { num_guaranteed_colonies >= 1 }
		hyperlane_jumps = { min = 1 max = 3 }
		initializer = "neighbor_t1_first_colony"
	}
}
