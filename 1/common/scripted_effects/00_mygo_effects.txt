# Keep the positive Stardust Analysis situation aligned with formal story
# completion flags. Exact progress prevents duplicate/compatibility events from
# advancing the bar more than once.
mygo_sync_stardust_analysis = {
	if = {
		limit = {
			is_ai = no
			has_origin = origin_mygo_eternal_return
			OR = {
				has_country_flag = mygo_earth_tomb_complete
				has_country_flag = mygo_stardust_analysis_complete
			}
		}
		if = {
			limit = {
				any_situation = { is_situation_type = situation_mygo_stardust_analysis }
			}
			every_situation = {
				limit = { is_situation_type = situation_mygo_stardust_analysis }
				set_situation_progress = 100
			}
		}
		else = {
			set_country_flag = mygo_stardust_analysis_complete
			if = {
				limit = { NOT = { has_modifier = mygo_stardust_analysis_complete } }
				add_modifier = { modifier = mygo_stardust_analysis_complete days = -1 }
			}
		}
	}
	else_if = {
		limit = {
			is_ai = no
			has_origin = origin_mygo_eternal_return
		}
		if = {
			limit = {
				NOT = {
					any_situation = { is_situation_type = situation_mygo_stardust_analysis }
				}
			}
			start_situation = { type = situation_mygo_stardust_analysis }
		}
		every_situation = {
			limit = { is_situation_type = situation_mygo_stardust_analysis }
			if = {
				limit = { owner = { has_country_flag = mygo_relic_popy_station } }
				set_situation_progress = 80
			}
			else_if = {
				limit = { owner = { has_country_flag = mygo_relic_record } }
				set_situation_progress = 60
			}
			else_if = {
				limit = { owner = { has_country_flag = mygo_relic_black_gold } }
				set_situation_progress = 40
			}
			else_if = {
				limit = { owner = { has_country_flag = mygo_relic_string } }
				set_situation_progress = 20
			}
			else = {
				set_situation_progress = 0
			}
		}
	}
}

mygo_spawn_earth_tomb_site = {
	save_event_target_as = mygo_earth_tomb_owner_country
	remove_country_flag = mygo_earth_tomb_site_spawned

	# Prefer the actual vanilla nuclear-wasteland Earth. Scan the galaxy instead
	# of trusting sol_system_earth, which can be overwritten by forced Sol starts.
	random_galaxy_planet = {
		limit = {
			has_carrier_flag = nuked_earth
			is_planet_class = pc_nuked
			NOT = { exists = archaeological_site }
		}
		create_archaeological_site = mygo_earth_tomb_site
		save_event_target_as = mygo_earth_tomb_planet
		event_target:mygo_earth_tomb_owner_country = { set_country_flag = mygo_earth_tomb_site_spawned }
	}

	# If vanilla generated a wild non-wasteland Sol, reuse and normalize that
	# system. This prevents this mod from adding a second wild Sol.
	if = {
		limit = { NOT = { has_country_flag = mygo_earth_tomb_site_spawned } }
		random_system = {
			limit = {
				has_star_flag = sol
				NOT = { has_star_flag = empire_home_system }
				any_system_planet = {
					OR = {
						has_carrier_flag = planet_earth
						has_carrier_flag = nuked_earth
					}
					NOT = { exists = archaeological_site }
				}
			}
			set_name = NAME_Sol
			set_star_flag = nuked_sol
			remove_star_flag = primitive_sol
			remove_star_flag = geocentric_sol
			remove_star_flag = suppress_archaeological_sites
			random_system_planet = {
				limit = {
					OR = {
						has_carrier_flag = planet_earth
						has_carrier_flag = nuked_earth
					}
					NOT = { exists = archaeological_site }
				}
				if = {
					limit = { exists = owner }
					destroy_colony = yes
				}
				change_pc = pc_nuked
				set_name = NAME_Earth
				remove_carrier_flag = planet_earth
				remove_carrier_flag = suppress_archaeological_sites
				remove_carrier_flag = ignore_startup_effect
				set_carrier_flag = nuked_earth
				prevent_anomaly = yes
				clear_deposits = yes
				create_archaeological_site = mygo_earth_tomb_site
				save_event_target_as = mygo_earth_tomb_planet
				event_target:mygo_earth_tomb_owner_country = { set_country_flag = mygo_earth_tomb_site_spawned }
			}
		}
	}

	# A damaged or mod-altered wild Sol may no longer have a usable Earth.
	# Add the story Earth inside that same system rather than spawning another Sol.
	if = {
		limit = {
			NOT = { has_country_flag = mygo_earth_tomb_site_spawned }
			any_system = {
				has_star_flag = sol
				NOT = { has_star_flag = empire_home_system }
			}
		}
		random_system = {
			limit = {
				has_star_flag = sol
				NOT = { has_star_flag = empire_home_system }
			}
			set_name = NAME_Sol
			set_star_flag = nuked_sol
			spawn_planet = {
				name = NAME_Earth
				class = pc_nuked
				orbit_distance = 25
				orbit_angle = 120
				size = 18
				init_effect = {
					set_carrier_flag = mygo_earth_tomb_fallback_planet
					prevent_anomaly = yes
					clear_deposits = yes
					create_archaeological_site = mygo_earth_tomb_site
					save_event_target_as = mygo_earth_tomb_planet
					event_target:mygo_earth_tomb_owner_country = { set_country_flag = mygo_earth_tomb_site_spawned }
				}
			}
		}
	}

	# With no Sol of any kind, create the exact vanilla nuclear-wasteland Sol.
	if = {
		limit = {
			NOT = { has_country_flag = mygo_earth_tomb_site_spawned }
			NOT = { any_system = { has_star_flag = sol } }
		}
		capital_scope = {
			solar_system = {
				spawn_system = {
					min_distance = 12
					max_distance = 30
					max_jumps = 2
					hyperlane = yes
					initializer = "special_init_04"
				}
				last_created_system = {
					random_system_planet = {
						limit = { has_carrier_flag = nuked_earth }
						create_archaeological_site = mygo_earth_tomb_site
						save_event_target_as = mygo_earth_tomb_planet
						event_target:mygo_earth_tomb_owner_country = { set_country_flag = mygo_earth_tomb_site_spawned }
					}
				}
			}
		}
	}

	# A forced empire-home Sol cannot be replaced safely. Add a separate story
	# Earth with MyGO-only flags so vanilla never treats it as another Sol/Earth.
	if = {
		limit = {
			NOT = { has_country_flag = mygo_earth_tomb_site_spawned }
			NOT = {
				any_system = {
					has_star_flag = sol
					NOT = { has_star_flag = empire_home_system }
				}
			}
		}
		capital_scope = {
			solar_system = {
				spawn_system = {
					min_distance = 12
					max_distance = 30
					max_jumps = 2
					hyperlane = yes
					initializer = "mygo_earth_tomb_fallback_system_initializer"
				}
				last_created_system = {
					random_system_planet = {
						limit = { has_carrier_flag = mygo_earth_tomb_fallback_planet }
						create_archaeological_site = mygo_earth_tomb_site
						save_event_target_as = mygo_earth_tomb_planet
						event_target:mygo_earth_tomb_owner_country = { set_country_flag = mygo_earth_tomb_site_spawned }
					}
				}
			}
		}
	}
	if = {
		limit = { has_country_flag = mygo_earth_tomb_site_spawned }
		create_point_of_interest = {
			id = mygo_earth_tomb_poi
			name = mygo_earth_tomb_poi
			desc = mygo_earth_tomb_poi_desc
			event_chain = mygo_earth_tomb_chain
			location = event_target:mygo_earth_tomb_planet
		}
		remove_country_flag = mygo_earth_tomb_site_spawned
	}
}
