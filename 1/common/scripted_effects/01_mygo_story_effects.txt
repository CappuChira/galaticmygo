mygo_story_ensure_internal_comms_contact = {
	if = {
		limit = {
			NOT = {
				any_country = {
					has_country_flag = mygo_internal_comms_contact
				}
			}
		}
		create_country = {
			name = "NAME_mygo_internal_comms"
			type = mygo_internal_comms
			auto_delete = no
			flag = {
				icon = {
					category = "special"
					file = "ai_01.dds"
				}
				background = {
					category = "backgrounds"
					file = "00_solid.dds"
				}
				colors = {
					"blue"
					"black"
					"null"
					"null"
				}
			}
			effect = {
				set_country_flag = mygo_internal_comms_contact
			}
		}
	}

	every_country = {
		limit = {
			has_country_flag = mygo_internal_comms_contact
		}
		if = {
			limit = { NOT = { exists = ruler } }
			# Custom-diplomacy contacts need a ruler-backed profile, as in vanilla Gray.
			create_leader = {
				class = official
				species = root
				name = NAME_mygo_internal_comms
				event_leader = yes
				skill = 1
				set_age = 18
				hide_age = yes
				randomize_traits = no
				skip_background_generation = yes
			}
			set_leader = last_created_leader
		}
		establish_communications_no_message = root
	}
}

mygo_story_ensure_fixed_star_seekers_civic = {
	if = {
		limit = {
			has_valid_civic = civic_star_seekers
			NOT = { has_valid_civic = civic_mygo_star_seekers }
		}
		force_remove_civic = civic_star_seekers
		force_add_civic = civic_mygo_star_seekers
	}
}

mygo_story_enable_starting_council_reform = {
	# The vanilla council has finished initializing by the time mygo.2 fires.
	# Clearing the cooldown lets the player immediately replace vanilla seats
	# with MyGO seats, or keep any vanilla positions they prefer.
	set_government_cooldown = no
	set_country_flag = mygo_starting_council_reform_enabled
}

mygo_story_fix_parallel_band_portraits = {
	every_country = {
		limit = {
			OR = {
				has_country_flag = mygo_popipa
				has_country_flag = mygo_morfonica
				has_country_flag = mygo_tsukinomori
			}
		}
		if = {
			limit = { has_country_flag = mygo_popipa }
			clear_ethos = yes
			country_add_ethic = ethic_fanatic_xenophile
			country_add_ethic = ethic_pacifist
			change_government = {
				authority = auth_dictatorial
				remove_invalid_civics = yes
				civics = {
					civic = civic_free_haven
					civic = civic_environmentalist
				}
			}
			set_ai_personality = federation_builders
			ruler = {
				set_name = PRESCRIPTED_ruler_name_kasumi
				change_leader_portrait = kasumi_leader_mygo_portrait
				set_immortal = yes
				set_age = 18
				set_leader_flag = mygo_parallel_fixed_ruler
				set_leader_flag = mygo_parallel_kasumi
			}
			}
			else_if = {
				limit = { has_country_flag = mygo_morfonica }
				clear_ethos = yes
				country_add_ethic = ethic_fanatic_spiritualist
				country_add_ethic = ethic_pacifist
				change_government = {
					authority = auth_dictatorial
					remove_invalid_civics = yes
					civics = {
						civic = civic_environmentalist
						civic = civic_beastmasters
					}
				}
				set_ai_personality = erudite_explorers
				ruler = {
					set_name = PRESCRIPTED_ruler_name_mashiro
				change_leader_portrait = mashiro_leader_mygo_portrait
				set_immortal = yes
				set_age = 18
				set_leader_flag = mygo_parallel_fixed_ruler
					set_leader_flag = mygo_parallel_mashiro
				}
			}
			else_if = {
				limit = { has_country_flag = mygo_tsukinomori }
				clear_ethos = yes
				country_add_ethic = ethic_authoritarian
				country_add_ethic = ethic_fanatic_materialist

				# Preserve the original Sakiko before corporate elections can replace her.
				if = {
					limit = {
						NOT = {
							any_owned_leader = { has_leader_flag = mygo_parallel_sakiko }
						}
					}
					ruler = {
						set_name = PRESCRIPTED_ruler_name_sakiko
						change_leader_portrait = sakiko_leader_mygo_portrait
						set_immortal = yes
						set_age = 18
						set_leader_flag = mygo_parallel_fixed_ruler
						set_leader_flag = mygo_parallel_sakiko
					}
				}

				if = {
					limit = { has_country_flag = mygo_tsukinomori_ave_mujica }
					if = {
						limit = {
							OR = {
								NOT = { has_authority = auth_dictatorial }
								NOT = { has_civic = civic_cutthroat_politics }
								NOT = { has_civic = civic_functional_architecture }
							}
						}
						change_government = {
							authority = auth_dictatorial
							remove_invalid_civics = yes
							civics = {
								civic = civic_cutthroat_politics
								civic = civic_functional_architecture
							}
						}
					}
					set_name = NAME_MyGO_AveMujica
				}
				else = {
					if = {
						limit = {
							OR = {
								NOT = { has_authority = auth_corporate }
								NOT = { has_civic = civic_indentured_assets }
								NOT = { has_civic = civic_corporate_crafters }
							}
						}
						change_government = {
							authority = auth_corporate
							remove_invalid_civics = yes
							civics = {
								civic = civic_indentured_assets
								civic = civic_corporate_crafters
							}
						}
					}
					if = {
						limit = { has_country_flag = mygo_tsukinomori_anon_tokyo }
						set_name = NAME_MyGO_AnonTokyo
					}
				}

				# Corporate elections may pick another CEO; restore the marked Sakiko only when needed.
				if = {
					limit = {
						any_owned_leader = { has_leader_flag = mygo_parallel_sakiko }
						ruler = {
							NOT = { has_leader_flag = mygo_parallel_sakiko }
						}
					}
					random_owned_leader = {
						limit = { has_leader_flag = mygo_parallel_sakiko }
						save_event_target_as = mygo_parallel_sakiko_ruler
					}
					set_leader = event_target:mygo_parallel_sakiko_ruler
				}
				set_ai_personality = ruthless_capitalists
				ruler = {
					set_name = PRESCRIPTED_ruler_name_sakiko
					change_leader_portrait = sakiko_leader_mygo_portrait
					set_immortal = yes
					set_age = 18
				set_leader_flag = mygo_parallel_fixed_ruler
				set_leader_flag = mygo_parallel_sakiko
			}
			if = {
				limit = {
					NOT = { has_country_flag = mygo_story_mutsumi_recruited }
					NOT = {
						any_owned_leader = { has_leader_flag = mygo_story_mutsumi }
					}
				}
				create_leader = {
					class = official
					species = owner_main_species
					name = NAME_mygo_mutsumi
					skill = 10
					immortal = yes
					set_age = 18
					hide_age = yes
					event_leader = yes
					randomize_traits = no
					skip_background_generation = yes
					background_job = entertainer
					background_ethic = ethic_materialist
					gender = female
					effect = {
						change_leader_portrait = mutsumi_leader_mygo_portrait
						set_leader_tier = leader_tier_legendary
						set_leader_flag = mygo_story_mutsumi
						set_leader_flag = mygo_legendary_leader
						set_leader_flag = legendary_leader
						set_leader_flag = immune_to_negative_traits
					}
				}
			}
			if = {
				limit = {
					any_owned_leader = { has_leader_flag = mygo_story_mutsumi }
				}
				set_country_flag = mygo_story_mutsumi_recruited
			}
			every_owned_leader = {
				limit = { has_leader_flag = mygo_story_mutsumi }
				change_leader_portrait = mutsumi_leader_mygo_portrait
				set_leader_tier = leader_tier_legendary
				set_leader_flag = immune_to_negative_traits
			}
		}
	}
}
