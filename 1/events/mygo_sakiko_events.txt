# Repairs global band targets and retries missed capital visits while the
# arkship is already parked in one of the three home systems.
country_event = {
	id = mygo.648
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_mygo_eternal_return
	}

	immediate = {
		mygo_refresh_parallel_band_event_targets = yes
		if = {
			limit = {
				NOT = { has_country_flag = mygo_visited_popipa_capital }
				any_owned_fleet = {
					is_arkship_fleet = yes
					exists = solar_system
					solar_system = {
						has_star_flag = mygo_parallel_popipa_system
					}
				}
			}
			random_owned_fleet = {
				limit = {
					is_arkship_fleet = yes
					exists = solar_system
					solar_system = {
						has_star_flag = mygo_parallel_popipa_system
					}
				}
				fleet_event = { id = mygo.637 }
			}
		}
		if = {
			limit = {
				NOT = { has_country_flag = mygo_visited_morfonica_capital }
				any_owned_fleet = {
					is_arkship_fleet = yes
					exists = solar_system
					solar_system = {
						has_star_flag = mygo_parallel_morfonica_system
					}
				}
			}
			random_owned_fleet = {
				limit = {
					is_arkship_fleet = yes
					exists = solar_system
					solar_system = {
						has_star_flag = mygo_parallel_morfonica_system
					}
				}
				fleet_event = { id = mygo.638 }
			}
		}
		if = {
			limit = {
				NOT = { has_country_flag = mygo_visited_tsukinomori_capital }
				any_owned_fleet = {
					is_arkship_fleet = yes
					exists = solar_system
					solar_system = {
						has_star_flag = mygo_parallel_tsukinomori_system
					}
				}
			}
			random_owned_fleet = {
				limit = {
					is_arkship_fleet = yes
					exists = solar_system
					solar_system = {
						has_star_flag = mygo_parallel_tsukinomori_system
					}
				}
				fleet_event = { id = mygo.639 }
			}
		}
	}
}

# Monthly/explicit compatibility router. The timed flag lets the route retry if
# diplomacy changes during the 35-day delay.
country_event = {
	id = mygo.649
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_mygo_eternal_return
		has_country_flag = mygo_earth_tomb_complete
		has_country_flag = mygo_renamed_to_mygo
		has_country_flag = mygo_visited_tsukinomori_capital
		NOT = { has_country_flag = mygo_story_650_seen }
		NOT = { has_country_flag = mygo_sakiko_route_pending }
		any_country = {
			has_country_flag = mygo_tsukinomori
			is_country_type = default
			has_communications = root
			is_subject = no
			NOT = { is_at_war_with = root }
		}
	}

	immediate = {
		mygo_refresh_parallel_band_event_targets = yes
		set_timed_country_flag = {
			flag = mygo_sakiko_route_pending
			days = 45
		}
		country_event = { id = mygo.650 days = 35 }
	}
}

# The Tsukinomori board weighs the ultimatum against MyGO's fleet strength.
# ROOT = Tsukinomori, FROM = MyGO.

	id = mygo.682
	title = mygo_story.673.name
	desc = mygo_story.673.desc
	picture = GFX_evt_nomad_default
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		exists = from
		from = { has_origin = origin_mygo_eternal_return }
	}

	# Submit. A pathetic Tsukinomori always yields; an inferior one usually does.
	option = {
		name = mygo_story.672.a
		ai_chance = {
			factor = 35
			modifier = {
				factor = 100
				relative_power = {
					who = from
					category = fleet
					value = pathetic
				}
			}
			modifier = {
				factor = 4
				relative_power = {
					who = from
					category = fleet
					value = inferior
				}
			}
			modifier = {
				factor = 0.25
				relative_power = {
					who = from
					category = fleet
					value = superior
				}
			}
			modifier = {
				factor = 0
				relative_power = {
					who = from
					category = fleet
					value = overwhelming
				}
			}
		}
		from = {
			set_country_flag = mygo_tsukinomori_double_pressure_surrender
			country_event = { id = mygo.672 }
		}
	}

	# Refuse. This preserves the existing warning and 30-day mobilization window.
	option = {
		name = mygo_story.673.a
		ai_chance = {
			factor = 65
			modifier = {
				factor = 0
				relative_power = {
					who = from
					category = fleet
					value = pathetic
				}
			}
			modifier = {
				factor = 0.5
				relative_power = {
					who = from
					category = fleet
					value = inferior
				}
			}
			modifier = {
				factor = 2
				relative_power = {
					who = from
					category = fleet
					value = superior
				}
			}
			modifier = {
				factor = 100
				relative_power = {
					who = from
					category = fleet
					value = overwhelming
				}
			}
		}
		from = { country_event = { id = mygo.673 } }
	}
}
