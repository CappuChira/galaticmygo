# MyGO Live output supplement. The vanilla approach contributes +100%; this
# timed modifier adds +900% so the combined job output bonus is +1000%.
ship_event = {
	id = mygo.8
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		owner? = {
			has_valid_civic = civic_mygo_star_seekers
		}
	}

	immediate = {
		owner = {
			add_modifier = {
				modifier = mygo_live_output_boost
				days = 180
			}
		}
	}
}

country_event = {
	id = mygo.6
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		is_country_type = default
		can_issue_contracts_trigger = yes
		count_issued_contracts = 0
		NOT = { has_country_flag = mygo_contract_active }
		NOT = { has_country_flag = mygo_tour_contract_recently_issued }
		NOT = {
			any_country = {
				OR = {
					has_country_flag = mygo_tour_contract_open
					AND = {
						has_origin = origin_mygo_eternal_return
						has_event_chain = contract_mygo_tour_chain
					}
				}
			}
		}
		any_relation = {
			has_origin = origin_mygo_eternal_return
			is_nomadic = yes
			has_country_flag = mygo_tour_unlocked
			has_communications = prev
			NOT = { is_at_war_with = prev }
		}
		any_owned_planet = {
			is_colony = yes
			is_under_colonization = no
			planet = {
				NOT = { has_planet_flag = mygo_tour_visited }
			}
			OR = {
				is_capital = yes
				pop_amount >= 200
			}
		}
	}

	immediate = {
		random_list = {
			1 = {}
			999 = {
				modifier = {
					factor = 3
					has_country_flag = mygo_tsukinomori
				}
				modifier = {
					factor = 2
					OR = {
						has_country_flag = mygo_popipa
						has_country_flag = mygo_morfonica
					}
				}
				random_relation = {
					limit = {
						has_origin = origin_mygo_eternal_return
						is_nomadic = yes
						has_country_flag = mygo_tour_unlocked
						has_communications = prev
						NOT = { is_at_war_with = prev }
					}
					save_event_target_as = mygo_tour_contract_operator
				}
				if = {
					limit = { exists = event_target:mygo_tour_contract_operator }
					if = {
						limit = {
							any_owned_planet = {
								is_colony = yes
								is_under_colonization = no
								is_capital = yes
								planet = {
									NOT = { has_planet_flag = mygo_tour_visited }
								}
								solar_system = {
									event_target:mygo_tour_contract_operator = {
										contract_target_system_reachable = { SYSTEM = prev }
									}
								}
							}
						}
						ordered_owned_planet = {
							limit = {
								is_colony = yes
								is_under_colonization = no
								is_capital = yes
								planet = {
									NOT = { has_planet_flag = mygo_tour_visited }
								}
								solar_system = {
									event_target:mygo_tour_contract_operator = {
										contract_target_system_reachable = { SYSTEM = prev }
									}
								}
							}
							position = 0
							order_by = trigger:pop_amount
							save_event_target_as = mygo_tour_contract_target_planet
							solar_system = { save_event_target_as = mygo_tour_contract_target_system }
						}
					}
					else = {
						ordered_owned_planet = {
							limit = {
								is_colony = yes
								is_under_colonization = no
								pop_amount >= 200
								planet = {
									NOT = { has_planet_flag = mygo_tour_visited }
								}
								solar_system = {
									event_target:mygo_tour_contract_operator = {
										contract_target_system_reachable = { SYSTEM = prev }
									}
								}
							}
							position = 0
							order_by = trigger:pop_amount
							save_event_target_as = mygo_tour_contract_target_planet
							solar_system = { save_event_target_as = mygo_tour_contract_target_system }
						}
					}
					if = {
						limit = {
							exists = event_target:mygo_tour_contract_target_system
							event_target:mygo_tour_contract_operator = {
								contract_target_system_reachable = { SYSTEM = event_target:mygo_tour_contract_target_system }
							}
						}
						issue_contract = {
							contract = contract_mygo_tour
							location = event_target:mygo_tour_contract_target_system
						}
						if = {
							limit = {
								is_galactic_community_member = yes
								OR = {
									is_active_resolution = "resolution_mygo_galactic_tour_planet_initiative"
									is_active_resolution = "resolution_mygo_galactic_tour_mutual_benefit"
									is_active_resolution = "resolution_mygo_galactic_tour_public_stage_fund"
									is_active_resolution = "resolution_mygo_galactic_tour_universal_mandate"
								}
							}
							set_timed_country_flag = {
								flag = mygo_tour_contract_recently_issued
								days = 1200
							}
						}
						else = {
							set_timed_country_flag = {
								flag = mygo_tour_contract_recently_issued
								days = 1800
							}
						}
					}
				}
			}
		}
	}
}
